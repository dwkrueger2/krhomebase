audit_log

Application audit trail for CRUD operations and sensitive actions: actor, target, JSON deltas, request correlation, source, resource, idempotency key, and error metadata.

Raw SQL
CREATE TABLE audit_log (
    id                 TEXT PRIMARY KEY,
    actor_user_id      TEXT,
    action             TEXT NOT NULL,
    target_table       TEXT NOT NULL,
    target_id          TEXT NOT NULL,
    matter_id          TEXT,
    client_id          TEXT,
    ip_address         TEXT,
    user_agent         TEXT,
    previous_json      TEXT,
    new_json           TEXT,
    request_id         TEXT,
    actor_email        TEXT,
    actor_role         TEXT,
    source             TEXT,
    resource           TEXT,
    idempotency_key    TEXT,
    success            INTEGER NOT NULL DEFAULT 1 CHECK (success IN (0,1)),
    error_message      TEXT,
    created_at         TEXT NOT NULL DEFAULT (datetime('now')),
    FOREIGN KEY (actor_user_id) REFERENCES user(id),
    FOREIGN KEY (matter_id)     REFERENCES matter(id),
    FOREIGN KEY (client_id)     REFERENCES client(id)
);
