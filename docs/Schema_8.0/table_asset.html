Constraints
  • Primary key: id
  • Unique: matter_id, asset_number must be unique as a combination.
  • Foreign key: matter_id → matter(id) — no ON DELETE action declared (default NO ACTION)
  • Check: asset_category must be one of financial_account, retirement, real_estate, insurance, vehicle, entity_interest, tangible, digital_other.
  • Check: planning_status must be one of not_reviewed, action_needed, in_progress, completed, no_action_needed.

Raw SQL
CREATE TABLE asset (
    id               TEXT PRIMARY KEY,
    matter_id        TEXT NOT NULL,
    asset_number     TEXT NOT NULL,
    asset_type       TEXT NOT NULL,
    asset_category   TEXT NOT NULL
                         CHECK (asset_category IN (
                             'financial_account','retirement','real_estate',
                             'insurance','vehicle','entity_interest',
                             'tangible','digital_other')),
    display_name     TEXT NOT NULL,
    description      TEXT,
    planning_status  TEXT NOT NULL DEFAULT 'not_reviewed'
                         CHECK (planning_status IN (
                             'not_reviewed','action_needed','in_progress',
                             'completed','no_action_needed')),
    notes            TEXT,
    created_at       TEXT NOT NULL DEFAULT (datetime('now')),
    updated_at       TEXT,
    FOREIGN KEY (matter_id) REFERENCES matter(id),
    UNIQUE (matter_id, asset_number)
);
