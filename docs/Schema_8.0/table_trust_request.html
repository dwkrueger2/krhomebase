trust_request

A trust or retainer request for a matter and client: requested amount, due date, status, payment context, and the ledger entry that satisfied it.

  • Human-visible numbering column: request_number.

Raw SQL
CREATE TABLE trust_request (
  id TEXT PRIMARY KEY,
  matter_id TEXT NOT NULL,
  client_id TEXT,
  request_number TEXT NOT NULL,
  requested_amount REAL NOT NULL CHECK (requested_amount >= 0),
  request_date TEXT NOT NULL,
  due_date TEXT,
  status TEXT NOT NULL DEFAULT 'draft' CHECK (status IN ('draft','sent','partially_paid','paid','cancelled','void')),
  memo TEXT,
  created_by_user_id TEXT,
  created_at TEXT NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ','now')),
  updated_at TEXT NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ','now')),
  FOREIGN KEY (matter_id) REFERENCES matter(id) ON DELETE CASCADE,
  FOREIGN KEY (client_id) REFERENCES client(id) ON DELETE SET NULL,
  FOREIGN KEY (created_by_user_id) REFERENCES user(id) ON DELETE SET NULL
);
