matter_staff_assignment

History of which staff work a matter and in what role (responsible attorney, acting attorney, assisting staff, originating attorney). Rows are imported from an external practice-management system or created in KR Homebase.

Raw SQL
CREATE TABLE matter_staff_assignment (
    id                   TEXT PRIMARY KEY,
    matter_id            TEXT NOT NULL,
    user_id              TEXT,
    assignment_role      TEXT NOT NULL CHECK (assignment_role IN ('primary_attorney','responsible','acting','assisting','originating_attorney','billing','other')),
    staff_code           TEXT,
    staff_name_snapshot  TEXT,
    assigned_date        TEXT,
    ended_date           TEXT,
    is_current           INTEGER NOT NULL DEFAULT 1 CHECK (is_current IN (0,1)),
    source_system        TEXT,
    notes                TEXT,
    created_at           TEXT NOT NULL DEFAULT (datetime('now')),
    updated_at           TEXT,
    FOREIGN KEY (matter_id) REFERENCES matter(id),
    FOREIGN KEY (user_id)   REFERENCES user(id)
);
