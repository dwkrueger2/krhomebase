individual_address

Dated address assignment for an individual: which address, what type (home, mailing, work), whether it is the primary one, and when it was valid.

Domain: Contacts, Addresses, and Communication Methods

Raw SQL
CREATE TABLE individual_address (
    id                       TEXT PRIMARY KEY,
    individual_id            TEXT NOT NULL,
    address_id               TEXT NOT NULL,
    address_type_lookup_id   TEXT,
    use_context              TEXT NOT NULL DEFAULT 'general'
                                 CHECK (use_context IN (
                                     'general','client_contact','billing','employee_hr',
                                     'user_profile','emergency_contact','other')),
    is_primary               INTEGER NOT NULL DEFAULT 0 CHECK (is_primary IN (0,1)),
    valid_from               TEXT,
    valid_to                 TEXT,
    source                   TEXT,
    notes                    TEXT,
    created_at               TEXT NOT NULL DEFAULT (datetime('now')),
    updated_at               TEXT,
    FOREIGN KEY (individual_id)          REFERENCES individual(id),
    FOREIGN KEY (address_id)             REFERENCES address(id),
    FOREIGN KEY (address_type_lookup_id) REFERENCES lookup(id)
);
