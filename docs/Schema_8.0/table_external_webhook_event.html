external_webhook_event

Journal of inbound webhook events for idempotency, replay safety, error tracking, and forensic review. Unique per provider and external event id.

Raw SQL
CREATE TABLE external_webhook_event (
  id TEXT PRIMARY KEY,
  integration_account_id TEXT,
  provider TEXT NOT NULL,
  external_event_id TEXT,
  event_type TEXT NOT NULL,
  resource_type TEXT,
  resource_id TEXT,
  received_at TEXT NOT NULL,
  processed_at TEXT,
  processing_status TEXT NOT NULL DEFAULT 'received' CHECK (processing_status IN ('received','processed','ignored','failed')),
  payload_json TEXT,
  error_message TEXT,
  created_at TEXT NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ','now')),
  FOREIGN KEY (integration_account_id) REFERENCES external_integration_account(id) ON DELETE SET NULL
);
