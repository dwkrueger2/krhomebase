Common usage notes
  • Controlled values are enforced by CHECK constraints on: billing_status. Update the constraint (table rebuild in SQLite/D1) when adding a value.
  • Money columns (unit_cost, amount) are stored as REAL (floating point). Round at the application layer; consider integer cents for new money columns.
Raw SQL
CREATE TABLE expense_entry (
  id TEXT PRIMARY KEY,
  matter_id TEXT NOT NULL,
  user_id TEXT,
  expense_date TEXT NOT NULL,
  expense_category TEXT NOT NULL,
  description TEXT,
  quantity REAL NOT NULL DEFAULT 1,
  unit_cost REAL NOT NULL DEFAULT 0,
  amount REAL NOT NULL DEFAULT 0,
  billing_status TEXT NOT NULL DEFAULT 'unbilled' CHECK (billing_status IN ('unbilled','billed','non_billable','written_off')),
  invoice_id TEXT,
  source_system TEXT,
  source_record_id TEXT,
  created_at TEXT NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ','now')),
  updated_at TEXT NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ','now')),
  FOREIGN KEY (matter_id) REFERENCES matter(id) ON DELETE CASCADE,
  FOREIGN KEY (user_id) REFERENCES user(id) ON DELETE SET NULL,
  FOREIGN KEY (invoice_id) REFERENCES invoice(id) ON DELETE SET NULL
);
