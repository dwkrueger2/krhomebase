user_role_set

Assignment of role sets to users with a primary flag, active flag, and optional start and expiry dates. Replaces the single role_set_id on user for multi-role users.

Raw SQL
CREATE TABLE user_role_set (
    id                    TEXT PRIMARY KEY,
    user_id               TEXT NOT NULL,
    role_set_id           TEXT NOT NULL,
    is_primary            INTEGER NOT NULL DEFAULT 0 CHECK (is_primary IN (0,1)),
    is_active             INTEGER NOT NULL DEFAULT 1 CHECK (is_active IN (0,1)),
    assigned_by_user_id   TEXT,
    assigned_at           TEXT NOT NULL DEFAULT (datetime('now')),
    starts_at             TEXT,
    expires_at            TEXT,
    created_at            TEXT NOT NULL DEFAULT (datetime('now')),
    updated_at            TEXT,
    FOREIGN KEY (user_id)             REFERENCES user(id) ON DELETE CASCADE,
    FOREIGN KEY (role_set_id)         REFERENCES role_set(id) ON DELETE CASCADE,
    FOREIGN KEY (assigned_by_user_id) REFERENCES user(id) ON DELETE SET NULL,
    UNIQUE (user_id, role_set_id)
);
