public_inquiry_stage_history

History of review-stage changes for an inquiry: the previous status, the new status, who changed it, and when. Added in v2.0.

CREATE TABLE public_inquiry_stage_history (
  id TEXT PRIMARY KEY,
  inquiry_id TEXT NOT NULL,
  previous_status TEXT,
  next_status TEXT NOT NULL,
  event_note TEXT,
  created_by TEXT NOT NULL,
  created_at TEXT NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ','now')),
  FOREIGN KEY (inquiry_id) REFERENCES public_inquiry(id) ON DELETE CASCADE
);
